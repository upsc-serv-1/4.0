```typescript
// src/store/slices/unifiedNoteSlice.ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit';

interface FormattingState {
  highlightColor: string | null;
  isUnderlined: boolean;
  isStrikethrough: boolean;
  isBold: boolean;
  isItalic: boolean;
  textColor: string;
}

interface NoteState {
  id: string;
  text: string;
  formatting: FormattingState;
  parentId: string | null;
  children: NoteState[];
  isLocked: boolean;
  source?: string;
}

const unifiedNoteSlice = createSlice({
  name: 'unifiedNotes',
  initialState: {
    byId: {} as Record<string, NoteState>,
    allIds: [] as string[],
    byNotebook: {} as Record<string, string[]>,
    selectedNoteId: null as string | null,
    editingNoteId: null as string | null,
  },
  reducers: {
    // Add/update note
    upsertNote: (state, action: PayloadAction<NoteState>) => {
      const note = action.payload;
      state.byId[note.id] = note;
      if (!state.allIds.includes(note.id)) {
        state.allIds.push(note.id);
      }
    },

    // Update formatting
    updateFormatting: (state, action: PayloadAction<{
      noteId: string;
      formatting: Partial<FormattingState>;
    }>) => {
      const { noteId, formatting } = action.payload;
      if (state.byId[noteId]) {
        state.byId[noteId].formatting = {
          ...state.byId[noteId].formatting,
          ...formatting,
        };
      }
    },

    // Update text
    updateNoteText: (state, action: PayloadAction<{
      noteId: string;
      text: string;
    }>) => {
      const { noteId, text } = action.payload;
      if (state.byId[noteId]) {
        state.byId[noteId].text = text;
      }
    },

    // Add child note
    addChildNote: (state, action: PayloadAction<{
      parentId: string;
      childNote: NoteState;
    }>) => {
      const { parentId, childNote } = action.payload;
      state.byId[childNote.id] = childNote;
      if (!state.allIds.includes(childNote.id)) {
        state.allIds.push(childNote.id);
      }
      
      if (state.byId[parentId]) {
        state.byId[parentId].children.push(childNote);
      }
    },

    // Delete note
    deleteNote: (state, action: PayloadAction<string>) => {
      const noteId = action.payload;
      delete state.byId[noteId];
      state.allIds = state.allIds.filter(id => id !== noteId);
    },

    // Select for viewing
    selectNote: (state, action: PayloadAction<string>) => {
      state.selectedNoteId = action.payload;
    },

    // Start editing
    startEditingNote: (state, action: PayloadAction<string>) => {
      state.editingNoteId = action.payload;
    },

    // Stop editing
    stopEditingNote: (state) => {
      state.editingNoteId = null;
    },
  },
});

export const {
  upsertNote, updateFormatting, updateNoteText,
  addChildNote, deleteNote, selectNote,
  startEditingNote, stopEditingNote,
} = unifiedNoteSlice.actions;

export default unifiedNoteSlice.reducer;
```

## PHASE 3: SYNC BETWEEN TABS (Week 3-4)

### Goal: Make changes in Hard Notes appear in Notes tab (and vice versa)

### 3.1 Create Sync Service

```typescript
// src/lib/sync/UnifiedNoteSync.ts
import { AppDispatch } from '@/store';
import { upsertNote, deleteNote } from '@/store/slices/unifiedNoteSlice';
import { supabase } from '@/lib/supabase';

export class UnifiedNoteSync {
  constructor(
    private dispatch: AppDispatch,
    private userId: string
  ) {}

  /**
   * When user edits note in Hard Notes, sync to database
   */
  async syncNoteEdit(noteId: string, text: string, formatting: any) {
    // Update in Redux (instant UI update)
    this.dispatch(upsertNote({
      id: noteId,
      text,
      formatting,
      // ... other fields
    }));

    // Update in Supabase (background)
    try {
      await supabase
        .from('user_notes')
        .update({
          text,
          is_highlighted: !!formatting.highlightColor,
          highlight_color: formatting.highlightColor,
          is_underlined: formatting.isUnderlined,
          is_strikethrough: formatting.isStrikethrough,
          edited_at: new Date().toISOString(),
        })
        .eq('id', noteId)
        .eq('user_id', this.userId);
    } catch (error) {
      console.error('Sync error:', error);
      // Retry logic here
    }
  }

  /**
   * When user deletes note in Hard Notes, sync to database
   */
  async syncNoteDeletion(noteId: string) {
    this.dispatch(deleteNote(noteId));

    try {
      await supabase
        .from('user_notes')
        .delete()
        .eq('id', noteId)
        .eq('user_id', this.userId);
    } catch (error) {
      console.error('Delete sync error:', error);
    }
  }

  /**
   * Load all notes for a notebook
   */
  async loadNotebookNotes(notebookId: string) {
    const { data, error } = await supabase
      .from('user_notes')
      .select('*')
      .eq('notebook_id', notebookId)
      .eq('user_id', this.userId)
      .order('sort_order', { ascending: true });

    if (error) throw error;

    // Update Redux with all notes
    data.forEach(note => {
      this.dispatch(upsertNote({
        id: note.id,
        text: note.text,
        formatting: {
          highlightColor: note.highlight_color,
          isUnderlined: note.is_underlined,
          isStrikethrough: note.is_strikethrough,
        },
        parentId: note.parent_note_id,
        isLocked: note.is_locked,
        // ... map other fields
      }));
    });

    return data;
  }

  /**
   * Subscribe to real-time changes
   * (if user edits in Notes tab, Hard Notes updates automatically)
   */
  subscribeToChanges(notebookId: string) {
    const subscription = supabase
      .from(`user_notes:notebook_id=eq.${notebookId}`)
      .on('*', payload => {
        const { eventType, new: newRecord, old: oldRecord } = payload;

        switch (eventType) {
          case 'INSERT':
          case 'UPDATE':
            this.dispatch(upsertNote({
              id: newRecord.id,
              text: newRecord.text,
              formatting: {
                highlightColor: newRecord.highlight_color,
                isUnderlined: newRecord.is_underlined,
                isStrikethrough: newRecord.is_strikethrough,
              },
              parentId: newRecord.parent_note_id,
              isLocked: newRecord.is_locked,
            }));
            break;

          case 'DELETE':
            this.dispatch(deleteNote(oldRecord.id));
            break;
        }
      })
      .subscribe();

    return subscription;
  }
}
```

```typescript
// src/lib/export/PdfExporter.ts
import { PDFDocument, PDFPage } from 'react-native-pdf-lib';

export async function exportNotesToPDF(notes: NotePoint[], title: string) {
  const pdf = PDFDocument.create();

  // Add title page
  const page = pdf.addPage();
  page.fontSize(24)
    .text(title, 50, 50, { bold: true });

  // Add notes
  let yPosition = 100;
  notes.forEach(note => {
    const indent = (note.depth || 0) * 20;

    page.fontSize(12)
      .text(`${note.text}`, 50 + indent, yPosition);

    if (note.formatting?.highlightColor) {
      // Draw highlight background
      const colors: Record<string, string> = {
        yellow: '#FFFF00',
        green: '#00FF00',
        pink: '#FF1493',
        blue: '#0000FF',
      };
      page.fillColor(colors[note.formatting.highlightColor])
        .rect(40 + indent, yPosition - 5, 500, 20)
        .fill();
    }

    yPosition += 30;

    // Create new page if needed
    if (yPosition > 750) {
      yPosition = 50;
    }
  });

  // Save to device
  const path = `${RNFS.DocumentDirectoryPath}/notes-${Date.now()}.pdf`;
  await pdf.write(path);

  return path;
}
```

```typescript
export function exportNotesToMarkdown(notes: NotePoint[], title: string): string {
  let markdown = `# ${title}\n\n`;

  notes.forEach(note => {
    const indent = '  '.repeat(note.depth || 0);
    const bullet = '•';
    const text = note.text;

    let line = `${indent}${bullet} ${text}`;

    // Add formatting
    if (note.formatting?.highlightColor) {
      line = `> ${line}  (${note.formatting.highlightColor})`;
    }
    if (note.formatting?.isStrikethrough) {
      line = `${indent}${bullet} ~~${text}~~`;
    }

    markdown += `${line}\n`;
  });

  return markdown;
}
```
