### 2.2 Redux Slices

```typescript
// src/store/slices/strokeSlice.ts
import { createSlice, PayloadAction } from '@reduxjs/toolkit';

const strokeSlice = createSlice({
  name: 'strokes',
  initialState: {
    byId: {} as Record<string, Stroke>,
    allIds: [] as string[],
    historyStack: [] as Stroke[][],
    historyIndex: -1,
  },
  reducers: {
    addStroke: (state, action: PayloadAction<Stroke>) => {
      const stroke = action.payload;
      state.byId[stroke.id] = stroke;
      state.allIds.push(stroke.id);
      
      // Clear redo history on new action
      state.historyStack = state.historyStack.slice(0, state.historyIndex + 1);
      state.historyStack.push([...state.allIds]);
      state.historyIndex++;
    },

    updateStroke: (state, action: PayloadAction<{ id: string; patch: Partial<Stroke> }>) => {
      const { id, patch } = action.payload;
      if (state.byId[id]) {
        state.byId[id] = { ...state.byId[id], ...patch };
      }
    },

    deleteStroke: (state, action: PayloadAction<string>) => {
      const id = action.payload;
      delete state.byId[id];
      state.allIds = state.allIds.filter(sid => sid !== id);
    },

    deleteMultipleStrokes: (state, action: PayloadAction<string[]>) => {
      action.payload.forEach(id => {
        delete state.byId[id];
      });
      state.allIds = state.allIds.filter(id => !action.payload.includes(id));
    },

    undoStroke: (state) => {
      if (state.historyIndex > 0) {
        state.historyIndex--;
        const strokeIds = state.historyStack[state.historyIndex];
        
        // Reconstruct from history
        const newById: Record<string, Stroke> = {};
        strokeIds.forEach(id => {
          if (state.byId[id]) {
            newById[id] = state.byId[id];
          }
        });
        state.byId = newById;
        state.allIds = [...strokeIds];
      }
    },

    redoStroke: (state) => {
      if (state.historyIndex < state.historyStack.length - 1) {
        state.historyIndex++;
        const strokeIds = state.historyStack[state.historyIndex];
        
        const newById: Record<string, Stroke> = {};
        strokeIds.forEach(id => {
          if (state.byId[id]) {
            newById[id] = state.byId[id];
          }
        });
        state.byId = newById;
        state.allIds = [...strokeIds];
      }
    },

    selectStroke: (state, action: PayloadAction<string>) => {
      Object.values(state.byId).forEach(stroke => {
        stroke.isSelected = false;
      });
      if (state.byId[action.payload]) {
        state.byId[action.payload].isSelected = true;
      }
    },

    selectMultipleStrokes: (state, action: PayloadAction<string[]>) => {
      Object.values(state.byId).forEach(stroke => {
        stroke.isSelected = action.payload.includes(stroke.id);
      });
    },
  },
});

export const {
  addStroke, updateStroke, deleteStroke, deleteMultipleStrokes,
  undoStroke, redoStroke, selectStroke, selectMultipleStrokes,
} = strokeSlice.actions;

export default strokeSlice.reducer;
```

## 3. PENCIL INPUT HANDLER

### 3.1 Gesture Handler Integration

```typescript
// src/components/canvas/PencilInputManager.ts
import { GestureDetector, Gesture } from 'react-native-gesture-handler';
import Animated, { useSharedValue, withSpring, runOnJS } from 'react-native-reanimated';
import { AppDispatch } from '@/store';

export class PencilInputManager {
  private pressureData: Array<{
    x: number; y: number; pressure: number;
    tilt: number; angle: number; timestamp: number;
  }> = [];

  private startX: Animated.SharedValue<number>;
  private startY: Animated.SharedValue<number>;
  private currentStrokeId: string | null = null;

  constructor(private dispatch: AppDispatch) {
    this.startX = useSharedValue(0);
    this.startY = useSharedValue(0);
  }

  /**
   * Parse pencil event and extract pressure data
   */
  extractPencilProperties(event: any) {
    // event comes from React Native's native bridge
    
    const touch = event.nativeEvent.touches[0];
    if (!touch) return null;

    // Access Apple Pencil properties via native module
    return {
      x: touch.pageX,
      y: touch.pageY,
      pressure: touch.force ?? 0.5,                    // 0-1
      tilt: this.getTiltFromEvent(event),              // 0-360
      angle: this.getAltitudeFromEvent(event),         // 0-90
      timestamp: event.nativeEvent.timestamp,
      isPencil: touch.touchType === 'stylus',
    };
  }

  /**
   * Begin stroke: capture initial position
   */
  beginStroke(x: number, y: number, pencilProps: any) {
    this.startX.value = x;
    this.startY.value = y;
    this.pressureData = [];
    
    if (pencilProps) {
      this.pressureData.push({
        x, y,
        pressure: pencilProps.pressure,
        tilt: pencilProps.tilt,
        angle: pencilProps.angle,
        timestamp: pencilProps.timestamp,
      });
    }

    // Generate unique ID for this stroke
    this.currentStrokeId = `stroke_${Date.now()}_${Math.random()}`;
  }

  /**
   * Update stroke: accumulate pressure points
   */
  updateStroke(x: number, y: number, pencilProps: any) {
    if (pencilProps && this.currentStrokeId) {
      this.pressureData.push({
        x, y,
        pressure: pencilProps.pressure,
        tilt: pencilProps.tilt,
        angle: pencilProps.angle,
        timestamp: pencilProps.timestamp,
      });
    }
  }

  /**
   * End stroke: interpolate and add to Redux
   */
  endStroke(dispatch: any, toolState: any) {
    if (!this.currentStrokeId || this.pressureData.length < 2) return;

    const stroke = {
      id: this.currentStrokeId,
      pageId: getCurrentPageId(),
      toolType: toolState.tool,
      color: toolState.color,
      width: toolState.width,
      opacity: toolState.opacity,
      pressureData: this.pressureData,
      bezierPoints: this.interpolateStroke(this.pressureData),
      boundingBox: this.calculateBounds(this.pressureData),
      zIndex: getMaxZIndex() + 1,
      createdAt: Date.now(),
      updatedAt: Date.now(),
      isSelected: false,
      isLocked: false,
    };

    dispatch(addStroke(stroke));
    
    // Reset state
    this.currentStrokeId = null;
    this.pressureData = [];
  }

  /**
   * Interpolate raw pressure points into smooth Bezier curves
   */
  private interpolateStroke(points: any[]): BezierPoint[] {
    if (points.length < 3) {
      return points.map(p => ({
        x: p.x, y: p.y,
        cpX1: p.x, cpY1: p.y,
        cpX2: p.x, cpY2: p.y,
      }));
    }

    const result: BezierPoint[] = [];

    for (let i = 0; i < points.length - 1; i++) {
      const p0 = points[Math.max(0, i - 1)];
      const p1 = points[i];
      const p2 = points[i + 1];
      const p3 = points[Math.min(points.length - 1, i + 2)];

      // Catmull-Rom to Cubic Bezier conversion
      const cp1 = this.catmullRomControlPoint(p0, p1, p2, 0.5);
      const cp2 = this.catmullRomControlPoint(p1, p2, p3, 0.5);

      result.push({
        x: p1.x,
        y: p1.y,
        cpX1: cp1.x,
        cpY1: cp1.y,
        cpX2: cp2.x,
        cpY2: cp2.y,
        pressure: p1.pressure,
        velocity: this.calculateVelocity(p0, p1),
      });
    }

    // Add final point
    const lastP = points[points.length - 1];
    result.push({
      x: lastP.x,
      y: lastP.y,
      cpX1: lastP.x,
      cpY1: lastP.y,
      cpX2: lastP.x,
      cpY2: lastP.y,
      pressure: lastP.pressure,
      velocity: 0,
    });

    return result;
  }

  private catmullRomControlPoint(p0: any, p1: any, p2: any, t: number) {
    return {
      x: p1.x + (p2.x - p0.x) * t / 6,
      y: p1.y + (p2.y - p0.y) * t / 6,
    };
  }

  private calculateVelocity(p0: any, p1: any): number {
    const dx = p1.x - p0.x;
    const dy = p1.y - p0.y;
    const dt = (p1.timestamp - p0.timestamp) || 1;
    return Math.sqrt(dx * dx + dy * dy) / dt;
  }

  private calculateBounds(points: any[]) {
    let minX = Infinity, maxX = -Infinity;
    let minY = Infinity, maxY = -Infinity;

    points.forEach(p => {
      minX = Math.min(minX, p.x);
      maxX = Math.max(maxX, p.x);
      minY = Math.min(minY, p.y);
      maxY = Math.max(maxY, p.y);
    });

    return {
      x: minX,
      y: minY,
      width: maxX - minX,
      height: maxY - minY,
    };
  }

  private getTiltFromEvent(event: any): number {
    // Native module to extract azimuthAngle from UIEvent
    return NativeModules.PencilEventParser.getTilt(event.nativeEvent) ?? 0;
  }

  private getAltitudeFromEvent(event: any): number {
    return NativeModules.PencilEventParser.getAltitude(event.nativeEvent) ?? 45;
  }
}
```

### 3.2 Native Module for Pencil Properties

**PencilEventParser.swift (Objective-C++ bridge):**
```swift
import Foundation

@objc(PencilEventParser)
class PencilEventParser: NSObject {
  @objc static func getTilt(_ nativeEvent: [AnyHashable: Any]) -> NSNumber {
    // Access UIEvent from native event
    if let touch = nativeEvent["touch"] as? UITouch {
      return NSNumber(value: touch.azimuthAngle(in: nil))
    }
    return NSNumber(value: 0)
  }

  @objc static func getAltitude(_ nativeEvent: [AnyHashable: Any]) -> NSNumber {
    if let touch = nativeEvent["touch"] as? UITouch {
      return NSNumber(value: touch.altitudeAngle)
    }
    return NSNumber(value: 45) // degrees
  }
}
```

## 4. STROKE RENDERING WITH SKIA

### 4.1 Canvas Renderer Component

```typescript
// src/components/canvas/CanvasRenderer.tsx
import React, { useMemo, useCallback } from 'react';
import {
  Canvas, Path, Paint, Group, Skia,
  BlendMode, RoundedRect,
} from '@shopify/react-native-skia';
import { useSelector } from 'react-redux';
import Animated, { useAnimatedStyle } from 'react-native-reanimated';
import type { RootState } from '@/store';

interface CanvasRendererProps {
  canvasWidth: number;
  canvasHeight: number;
  zoom: number;
  panX: number;
  panY: number;
  rotation: number;
  currentStroke?: BezierPoint[];
  currentToolType: 'pen' | 'highlighter' | 'eraser' | 'shape';
  currentColor: string;
  currentWidth: number;
  currentOpacity: number;
}

export const CanvasRenderer = React.memo(({
  canvasWidth, canvasHeight,
  zoom, panX, panY, rotation,
  currentStroke,
  currentToolType, currentColor, currentWidth, currentOpacity,
}: CanvasRendererProps) => {
  const strokes = useSelector((state: RootState) =>
    state.strokes.allIds.map(id => state.strokes.byId[id]).filter(Boolean)
  );

  // Render stroke to Bezier path
  const strokePath = useCallback((stroke: Stroke) => {
    const path = Skia.Path.Make();

    if (stroke.bezierPoints.length === 0) return path;

    const firstPoint = stroke.bezierPoints[0];
    path.moveTo(firstPoint.x, firstPoint.y);

    for (let i = 0; i < stroke.bezierPoints.length - 1; i++) {
      const p1 = stroke.bezierPoints[i];
      const p2 = stroke.bezierPoints[i + 1];

      // Cubic Bezier curve
      path.cubicTo(
        p1.cpX2, p1.cpY2,
        p2.cpX1, p2.cpY1,
        p2.x, p2.y
      );
    }

    return path;
  }, []);

  // Create paint for stroke
  const createPaint = useCallback((stroke: Stroke) => {
    const paint = Skia.Paint();
    paint.setColor(Skia.Color(stroke.color));
    paint.setAlphaf(stroke.opacity);
    paint.setStrokeWidth(stroke.width);
    paint.setStrokeJoin('round');
    paint.setStrokeCap('round');
    paint.setStyle('stroke');

    // Different blend mode for highlighter
    if (stroke.toolType === 'highlighter') {
      paint.setBlendMode(BlendMode.Multiply);
      paint.setAlphaf(stroke.opacity * 0.3); // More transparent
    }

    return paint;
  }, []);

  // Animated canvas transformation
  const animatedStyle = useAnimatedStyle(() => ({
    transform: [
      { translateX: panX },
      { translateY: panY },
      { scale: zoom },
      { rotate: `${rotation}deg` },
    ],
  }));

  return (
    <Animated.View style={[{ width: canvasWidth, height: canvasHeight }, animatedStyle]}>
      <Canvas style={{ width: canvasWidth, height: canvasHeight }}>
        {/* White background */}
        <RoundedRect
          rect={Skia.XYWHRect(0, 0, canvasWidth, canvasHeight)}
          color="white"
        />

        {/* Grid background (optional) */}
        <GridBackground
          width={canvasWidth}
          height={canvasHeight}
          spacing={28}
          opacity={0.1}
        />

        {/* Render all strokes in Z-order */}
        <Group>
          {strokes.map((stroke) => {
            const path = strokePath(stroke);
            const paint = createPaint(stroke);

            return (
              <Path
                key={stroke.id}
                path={path}
                paint={paint}
                opacity={stroke.isSelected ? 0.7 : 1}
              />
            );
          })}

          {/* Currently drawing stroke (live) */}
          {currentStroke && currentStroke.length > 0 && (
            <LiveStroke
              points={currentStroke}
              toolType={currentToolType}
              color={currentColor}
              width={currentWidth}
              opacity={currentOpacity}
            />
          )}
        </Group>

        {/* Selection box if stroke selected */}
        {strokes.some(s => s.isSelected) && (
          <SelectionBox
            strokes={strokes.filter(s => s.isSelected)}
          />
        )}
      </Canvas>
    </Animated.View>
  );
});

// Live stroke during drawing
const LiveStroke = React.memo(({
  points, toolType, color, width, opacity,
}: {
  points: BezierPoint[];
  toolType: string;
  color: string;
  width: number;
  opacity: number;
}) => {
  if (points.length < 2) return null;

  const path = Skia.Path.Make();
  path.moveTo(points[0].x, points[0].y);

  for (let i = 0; i < points.length - 1; i++) {
    const p1 = points[i];
    const p2 = points[i + 1];
    path.cubicTo(p1.cpX2, p1.cpY2, p2.cpX1, p2.cpY1, p2.x, p2.y);
  }

  const paint = Skia.Paint();
  paint.setColor(Skia.Color(color));
  paint.setAlphaf(opacity);
  paint.setStrokeWidth(width);
  paint.setStrokeJoin('round');
  paint.setStrokeCap('round');
  paint.setStyle('stroke');

  if (toolType === 'highlighter') {
    paint.setBlendMode(BlendMode.Multiply);
  }

  return <Path path={path} paint={paint} />;
});

// Selection box around selected strokes
const SelectionBox = ({ strokes }: { strokes: Stroke[] }) => {
  const bounds = useMemo(() => {
    let minX = Infinity, maxX = -Infinity;
    let minY = Infinity, maxY = -Infinity;

    strokes.forEach(stroke => {
      minX = Math.min(minX, stroke.boundingBox.x);
      maxX = Math.max(maxX, stroke.boundingBox.x + stroke.boundingBox.width);
      minY = Math.min(minY, stroke.boundingBox.y);
      maxY = Math.max(maxY, stroke.boundingBox.y + stroke.boundingBox.height);
    });

    return { x: minX - 5, y: minY - 5, width: maxX - minX + 10, height: maxY - minY + 10 };
  }, [strokes]);

  const paint = Skia.Paint();
  paint.setColor(Skia.Color('#007AFF'));
  paint.setStrokeWidth(2);
  paint.setStyle('stroke');

  return (
    <RoundedRect
      x={bounds.x}
      y={bounds.y}
      width={bounds.width}
      height={bounds.height}
      r={6}
      paint={paint}
    />
  );
};

// Grid background helper
const GridBackground = ({ width, height, spacing, opacity }: any) => {
  const lines: React.ReactNode[] = [];
  const paint = Skia.Paint();
  paint.setColor(Skia.Color('#CCCCCC'));
  paint.setAlphaf(opacity);
  paint.setStrokeWidth(0.5);

  // Vertical lines
  for (let x = 0; x < width; x += spacing) {
    const path = Skia.Path.Make();
    path.moveTo(x, 0);
    path.lineTo(x, height);
    lines.push(<Path key={`v-${x}`} path={path} paint={paint} />);
  }

  // Horizontal lines
  for (let y = 0; y < height; y += spacing) {
    const path = Skia.Path.Make();
    path.moveTo(0, y);
    path.lineTo(width, y);
    lines.push(<Path key={`h-${y}`} path={path} paint={paint} />);
  }

  return <Group>{lines}</Group>;
};
```

## 5. CANVAS VIEWPORT & TRANSFORMS

### 5.1 Zoom & Pan Controller

```typescript
// src/hooks/useCanvasTransforms.ts
import { useSharedValue, useAnimatedStyle, withSpring } from 'react-native-reanimated';
import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import { useMemo } from 'react';

const MIN_ZOOM = 0.5;
const MAX_ZOOM = 5.0;
const ZOOM_SENSITIVITY = 0.05;

export function useCanvasTransforms() {
  const zoom = useSharedValue(1);
  const panX = useSharedValue(0);
  const panY = useSharedValue(0);
  const rotation = useSharedValue(0);

  // Pan gesture
  const panGesture = useMemo(
    () =>
      Gesture.Pan()
        .minDistance(10)
        .onUpdate((event) => {
          panX.value += event.velocityX * 0.01;
          panY.value += event.velocityY * 0.01;
        })
        .onEnd(() => {
          // Decelerate
          panX.value = withSpring(panX.value, {
            damping: 10,
            mass: 1,
            overshootClamping: true,
          });
          panY.value = withSpring(panY.value, {
            damping: 10,
            mass: 1,
            overshootClamping: true,
          });
        }),
    [panX, panY]
  );

  // Pinch zoom gesture
  const pinchGesture = useMemo(
    () =>
      Gesture.Pinch()
        .onUpdate((event) => {
          const newZoom = zoom.value * event.scale;
          if (newZoom >= MIN_ZOOM && newZoom <= MAX_ZOOM) {
            zoom.value = newZoom;
          }
        })
        .onEnd(() => {
          // Snap to common zoom levels
          const commonZooms = [0.5, 0.75, 1, 1.25, 1.5, 2, 3, 4, 5];
          const closest = commonZooms.reduce((prev, curr) =>
            Math.abs(curr - zoom.value) < Math.abs(prev - zoom.value) ? curr : prev
          );

          zoom.value = withSpring(closest, {
            damping: 8,
            mass: 1,
            overshootClamping: true,
          });
        }),
    [zoom]
  );

  // Rotation gesture (two-finger twist)
  const rotationGesture = useMemo(
    () =>
      Gesture.Rotation()
        .onUpdate((event) => {
          rotation.value = event.rotation;
        })
        .onEnd(() => {
          // Snap to 90-degree angles
          const snappedRotation = Math.round(rotation.value / (Math.PI / 2)) * (Math.PI / 2);
          rotation.value = withSpring(snappedRotation, {
            damping: 8,
            mass: 1,
          });
        }),
    [rotation]
  );

  // Double-tap to reset zoom
  const doubleTapGesture = useMemo(
    () =>
      Gesture.Tap()
        .numberOfTaps(2)
        .onStart(() => {
          zoom.value = withSpring(1, { damping: 8 });
          panX.value = withSpring(0, { damping: 8 });
          panY.value = withSpring(0, { damping: 8 });
          rotation.value = withSpring(0, { damping: 8 });
        }),
    [zoom, panX, panY, rotation]
  );

  const composed = Gesture.Simultaneous(
    pinchGesture,
```

## 6. SYNC ENGINE IMPLEMENTATION

### 6.1 CRDT Merge Algorithm

```typescript
// src/lib/crdt/merge.ts
export interface Operation {
  id: string;
  userId: string;
  timestamp: number;
  pageId: string;
  data: {
    type: 'ADD_STROKE' | 'DELETE_STROKE' | 'UPDATE_STROKE';
    strokeId?: string;
    stroke?: Stroke;
    patch?: Partial<Stroke>;
  };
}

export class CRDTMerge {
  /**
   * Merge remote operations with local state
   * Returns merged state that both clients can agree on
   */
  static merge(
    localState: Stroke[],
    localOps: Operation[],
    remoteOps: Operation[]
  ): Stroke[] {
    // 1. Create unified operation log
    const allOps = [...localOps, ...remoteOps]
      .sort((a, b) => {
        // Sort by timestamp first, then by userId for deterministic ordering
        if (a.timestamp !== b.timestamp) {
          return a.timestamp - b.timestamp;
        }
        return a.userId.localeCompare(b.userId);
      });

    // 2. Apply operations in order (commutative, order-independent)
    const finalState = new Map<string, Stroke>();

    // Initialize with current state
    localState.forEach(stroke => {
      finalState.set(stroke.id, { ...stroke });
    });

    // Apply each operation in chronological order
    for (const op of allOps) {
      switch (op.data.type) {
        case 'ADD_STROKE': {
          if (op.data.stroke) {
            finalState.set(op.data.stroke.id, op.data.stroke);
          }
          break;
        }

        case 'DELETE_STROKE': {
          finalState.delete(op.data.strokeId!);
          break;
        }

        case 'UPDATE_STROKE': {
          const existing = finalState.get(op.data.strokeId!);
          if (existing && op.data.patch) {
            finalState.set(op.data.strokeId!, {
              ...existing,
              ...op.data.patch,
            });
          }
          break;
        }
      }
    }

    return Array.from(finalState.values());
  }

  /**
   * Detect conflicts between local and remote operations
   */
  static detectConflicts(
    localOps: Operation[],
    remoteOps: Operation[]
  ): Conflict[] {
    const conflicts: Conflict[] = [];

    for (const localOp of localOps) {
      for (const remoteOp of remoteOps) {
        // Conflict: same stroke deleted locally and updated remotely
        if (
          localOp.data.type === 'DELETE_STROKE' &&
          remoteOp.data.type === 'UPDATE_STROKE' &&
          localOp.data.strokeId === remoteOp.data.strokeId
        ) {
          conflicts.push({
            type: 'DELETE_UPDATE_CONFLICT',
            localOp,
            remoteOp,
            resolution: 'FAVOR_LOCAL', // Delete wins
          });
        }

        // Conflict: concurrent updates to same stroke
        if (
          localOp.data.type === 'UPDATE_STROKE' &&
          remoteOp.data.type === 'UPDATE_STROKE' &&
          localOp.data.strokeId === remoteOp.data.strokeId &&
          Math.abs(localOp.timestamp - remoteOp.timestamp) < 1000 // Within 1 second
        ) {
          conflicts.push({
            type: 'CONCURRENT_UPDATE_CONFLICT',
            localOp,
            remoteOp,
            resolution: 'MERGE_PATCHES', // Merge both patches
          });
        }
      }
    }

    return conflicts;
  }
}

interface Conflict {
  type: string;
  localOp: Operation;
  remoteOp: Operation;
  resolution: string;
}
```

### 6.2 Sync Manager

```typescript
// src/lib/sync/SyncManager.ts
export class SyncManager {
  private syncQueue: Operation[] = [];
  private isSyncing = false;
  private lastSyncTime = 0;

  constructor(
    private db: Database,
    private supabase: SupabaseClient,
    private dispatch: AppDispatch,
    private getState: () => RootState
  ) {}

  /**
   * Record an operation locally and queue for sync
   */
  async recordOperation(op: Operation) {
    // Save to local database immediately
    await this.db.insert('operations', {
      ...op,
      synced_at: null,
    });

    this.syncQueue.push(op);

    // Try to sync if online
    if (this.isNetworkConnected()) {
      this.queueSync();
    }
  }

  /**
   * Perform full sync with server
   */
  async sync() {
    if (this.isSyncing) return;
    if (!this.isNetworkConnected()) return;

    this.isSyncing = true;
    this.dispatch(setSyncing(true));

    try {
      // 1. Get all unsynced local operations
      const unsyncedOps = await this.db.query(
        'SELECT * FROM operations WHERE synced_at IS NULL ORDER BY timestamp'
      );

      // 2. Send to server via RPC
      const { data, error } = await this.supabase.rpc('merge_operations', {
        operations: unsyncedOps,
        clientUserId: this.getState().auth.userId,
      });

      if (error) throw error;

      // 3. Get remote operations to apply
      const remoteOps = data.remoteOperations || [];
      const mergedState = CRDTMerge.merge(
        this.getCurrentPageStrokes(),
        unsyncedOps,
        remoteOps
      );

      // 4. Update local state
      this.dispatch(replaceStrokes(mergedState));

      // 5. Mark operations as synced
      await this.db.update(
        'operations',
        { synced_at: Date.now() },
        { synced_at: null }
      );

      this.lastSyncTime = Date.now();
      this.dispatch(setSyncError(null));
    } catch (error) {
      this.dispatch(setSyncError(error.message));
    } finally {
      this.isSyncing = false;
      this.dispatch(setSyncing(false));
    }
  }

  /**
   * Queue sync with debounce
   */
  private syncTimer: NodeJS.Timeout | null = null;

  private queueSync() {
    if (this.syncTimer) clearTimeout(this.syncTimer);

    this.syncTimer = setTimeout(() => {
      this.sync();
    }, 1000); // Debounce 1 second
  }

  private isNetworkConnected(): boolean {
    // Check network status
    return true; // Placeholder
  }

  private getCurrentPageStrokes(): Stroke[] {
    const state = this.getState();
    return state.strokes.allIds
      .map(id => state.strokes.byId[id])
      .filter(Boolean);
  }
}
```

```typescript
// src/hooks/useOptimisticUpdate.ts
export function useOptimisticUpdate() {
  const dispatch = useAppDispatch();

  return async <T extends any>(
    optimisticUpdate: () => void,
    serverUpdate: () => Promise<void>,
    onError?: (error: Error) => void
  ) => {
    // 1. Apply update optimistically
    optimisticUpdate();

    try {
      // 2. Sync with server
      await serverUpdate();
    } catch (error) {
      // 3. Rollback on error
      onError?.(error as Error);
      // Dispatch undo action or revert to previous state
    }
  };
}

  () => dispatch(removeStroke(newStroke.id))  // rollback
```

```typescript
// src/lib/db/BatchQuery.ts
export class BatchQuery {
  private batch: Operation[] = [];
  private timer: NodeJS.Timeout | null = null;

  add(op: Operation) {
    this.batch.push(op);
    this.scheduleFlush();
  }

  private scheduleFlush() {
    if (this.timer) clearTimeout(this.timer);

    this.timer = setTimeout(() => this.flush(), 100);
  }

  async flush() {
    if (this.batch.length === 0) return;

    const ops = this.batch.splice(0);
    
    // Send all at once
    await supabase.rpc('batch_operations', { operations: ops });
  }
}
```

```typescript
// Convert screen coordinates to canvas coordinates
function screenToCanvasCoords(
  screenX: number,
  screenY: number,
  zoom: number,
  panX: number,
  panY: number
) {
  return {
    x: (screenX - panX) / zoom,
    y: (screenY - panY) / zoom,
```
